# Enemy Active AI Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace `EmulatedGame`'s scattered enemy state with a symmetric `PlayerState` / `PlayerBehavior` architecture, introduce `EnemyStrategy` as a behaviour interface, a full three-race `TechTree`, named strategy library, and a reactive counter-picking strategy.

**Architecture:** `EmulatedGame` becomes a symmetric physics referee holding two `PlayerState` instances (friendly + enemy) and one `IntentQueue` per player. Both queues drain through the same generalized `applyIntent(Intent, PlayerState)`. `EnemyBehavior implements PlayerBehavior` and pushes `TrainIntent` / `BuildIntent` / `AttackIntent` into the enemy queue each tick. `AgentOrchestrator` continues filling the friendly queue unchanged.

**Tech Stack:** Java 21 records, sealed interfaces, JUnit 5 + AssertJ, Quarkus CDI (EmulatedEngine only), Drools (untouched), existing `IntentQueue` / `Intent` sealed hierarchy.

```java
public interface EnemyStrategy {
    String name();

}
```

```java
package io.quarkmind.sc2.emulated;

import io.quarkmind.domain.*;
import java.util.List;
import java.util.Optional;

public class FixedBuildOrderStrategy implements EnemyStrategy {

    private final String           name;
    private final Race             race;
    private final List<UnitType>   buildOrder;
    private final int              mineralsPerTick;
    private final EnemyAttackConfig attackConfig;
    private int buildIndex = 0;

    public FixedBuildOrderStrategy(String name, Race race, List<UnitType> buildOrder,
                                   int mineralsPerTick, EnemyAttackConfig attackConfig) {
        this.name            = name;
        this.race            = race;
        this.buildOrder      = List.copyOf(buildOrder);
        this.mineralsPerTick = mineralsPerTick;
        this.attackConfig    = attackConfig;
    }

    @Override public String           name()            { return name;            }
    @Override public Race             race()            { return race;            }
    @Override public int              mineralsPerTick() { return mineralsPerTick; }
    @Override public EnemyAttackConfig attackConfig()   { return attackConfig;    }

    @Override
    public Optional<UnitType> nextUnit(EnemyObservation obs) {
        if (buildOrder.isEmpty()) return Optional.empty();
        UnitType unit = buildOrder.get(buildIndex % buildOrder.size());
        buildIndex = (buildIndex + 1) % buildOrder.size();
        return Optional.of(unit);
    }

    @Override public boolean shouldSwitch(EnemyObservation obs) { return false; }

    @Override public void reset() { buildIndex = 0; }
}
```

```java
package io.quarkmind.sc2.emulated;

import io.quarkmind.domain.*;
import java.util.*;

public final class EnemyStrategyLibrary {

    private static final EnemyAttackConfig FAST_PUSH =
        new EnemyAttackConfig(4, 150, 25, 40);
    private static final EnemyAttackConfig SLOW_PUSH =
        new EnemyAttackConfig(6, 250, 20, 30);
    private static final EnemyAttackConfig FLOOD =
        new EnemyAttackConfig(8, 120, 15, 25);

    private static final Map<String, EnemyStrategy> REGISTRY = new LinkedHashMap<>();

    static {
        register(fixed("PROTOSS_4GATE",        Race.PROTOSS, FAST_PUSH, 2,
            UnitType.ZEALOT, UnitType.ZEALOT, UnitType.STALKER, UnitType.STALKER,
            UnitType.STALKER, UnitType.STALKER));
        register(fixed("PROTOSS_BLINK_STALKER", Race.PROTOSS, SLOW_PUSH, 2,
            UnitType.STALKER, UnitType.STALKER, UnitType.STALKER, UnitType.STALKER));
        register(fixed("PROTOSS_COLOSSUS_PUSH", Race.PROTOSS, SLOW_PUSH, 2,
            UnitType.ZEALOT, UnitType.IMMORTAL, UnitType.IMMORTAL, UnitType.COLOSSUS));
        register(fixed("ZERG_ROACH_RUSH",       Race.ZERG,    FAST_PUSH, 3,
            UnitType.ROACH, UnitType.ROACH, UnitType.ROACH,
            UnitType.ROACH, UnitType.ROACH, UnitType.ROACH));
        register(fixed("ZERG_MASS_LING",        Race.ZERG,    FLOOD,     3,
            UnitType.ZERGLING, UnitType.ZERGLING, UnitType.ZERGLING, UnitType.ZERGLING,
            UnitType.ZERGLING, UnitType.ZERGLING, UnitType.ZERGLING, UnitType.ZERGLING));
        register(fixed("ZERG_HYDRA_SWITCH",     Race.ZERG,    SLOW_PUSH, 3,
            UnitType.ZERGLING, UnitType.ZERGLING, UnitType.ZERGLING, UnitType.ZERGLING,
            UnitType.HYDRALISK, UnitType.HYDRALISK, UnitType.HYDRALISK, UnitType.HYDRALISK));
        register(fixed("TERRAN_2RAX_MARINE",    Race.TERRAN,  FLOOD,     2,
            UnitType.MARINE, UnitType.MARINE, UnitType.MARINE, UnitType.MARINE,
            UnitType.MARINE, UnitType.MARINE, UnitType.MARINE, UnitType.MARINE));
        register(fixed("TERRAN_BIO_PUSH",       Race.TERRAN,  FAST_PUSH, 2,
            UnitType.MARINE, UnitType.MARINE, UnitType.MARINE, UnitType.MARINE,
            UnitType.MARAUDER, UnitType.MARAUDER, UnitType.MEDIVAC));
        register(fixed("TERRAN_MECH",           Race.TERRAN,  SLOW_PUSH, 2,
            UnitType.HELLION, UnitType.HELLION, UnitType.SIEGE_TANK, UnitType.SIEGE_TANK));
        // REACTIVE: placeholder — replaced by ReactiveStrategy in Phase 3
        register(fixed("REACTIVE",              Race.PROTOSS, FAST_PUSH, 2,
            UnitType.ZEALOT, UnitType.STALKER));
    }

    private EnemyStrategyLibrary() {}

    private static FixedBuildOrderStrategy fixed(String name, Race race,
            EnemyAttackConfig atk, int mpt, UnitType... units) {
        return new FixedBuildOrderStrategy(name, race, List.of(units), mpt, atk);
    }

    private static void register(EnemyStrategy s) { REGISTRY.put(s.name(), s); }

    public static EnemyStrategy forName(String name) {
        EnemyStrategy s = REGISTRY.get(name);
        if (s == null) throw new IllegalArgumentException("Unknown strategy: " + name);
        // Return a fresh instance so callers don't share build index state
        if (s instanceof FixedBuildOrderStrategy f) {
            f.reset();
        }
        return s;
    }

    public static EnemyStrategy randomForRace(Race race) {
        List<EnemyStrategy> pool = REGISTRY.values().stream()
            .filter(s -> s.race() == race && !s.name().equals("REACTIVE"))
            .toList();
        if (pool.isEmpty()) throw new IllegalStateException("No strategies for race: " + race);
        return pool.get(new Random().nextInt(pool.size()));
    }

    public static Set<String> allNames() { return Collections.unmodifiableSet(REGISTRY.keySet()); }
}
```

```java
class PlayerState {

    record PendingCompletion(long completesAtTick, Runnable action) {}

    final List<Unit>               units               = new ArrayList<>();
    final List<Building>           buildings           = new ArrayList<>();
    final List<Unit>               stagingArea         = new ArrayList<>();
    final List<PendingCompletion>  pendingCompletions  = new ArrayList<>();

    int    vespene;
    int    supply;
    int    supplyUsed;

    void fireCompletions(long currentTick) {
        pendingCompletions.removeIf(item -> {
            if (item.completesAtTick() > currentTick) return false;
            item.action().run();
            return true;
        });
    }

    void clear() {
        units.clear();
        buildings.clear();
        stagingArea.clear();
        pendingCompletions.clear();
        minerals   = 0;
        vespene    = 0;
        supply     = 0;
        supplyUsed = 0;
```

```java
package io.quarkmind.sc2.emulated;

import io.quarkmind.domain.*;
import io.quarkmind.sc2.IntentQueue;
import org.jboss.logging.Logger;
import java.util.*;

public class EnemyBehavior implements PlayerBehavior {

    private static final Logger log = Logger.getLogger(EnemyBehavior.class);

    static final Point2d NEXUS_POS   = new Point2d(8, 8);
    static final Point2d STAGING_POS = new Point2d(26, 26);

    private final PlayerState          enemy;
    private       EnemyStrategy        strategy;
    private       Optional<UnitType>   currentTarget   = Optional.empty();
    private final Set<String>          retreatingUnits = new HashSet<>();
    private       int                  initialAttackSize;
    private       long                 framesSinceLastAttack;
    private       int                  nextTag         = 1000;

    public EnemyBehavior(EnemyStrategy strategy, PlayerState enemy) {
        this.strategy = strategy;
        this.enemy    = enemy;
    }

    @Override
    public void tick(GameState observation, IntentQueue queue) {
        long frame = observation.gameFrame();
        EnemyObservation obs = new EnemyObservation(
            observation.enemyUnits(),    // from enemy's POV these are our friendly units
            buildingTypes(),
            (int) enemy.minerals,
            frame
        );

        // 1. Accumulate minerals
        enemy.minerals += strategy.mineralsPerTick();

        // 2. Production
        tickProduction(obs, queue);

        // 3. Attack launch
        framesSinceLastAttack++;
        tickAttackLaunch(queue);

        // 4. Retreat
        tickRetreat(queue);

        // 5. Strategy switch check (Reactive — no-op for Fixed)
        if (strategy.shouldSwitch(obs)) {
            EnemyStrategy next = EnemyStrategyLibrary.randomForRace(strategy.race());
            log.infof("[ENEMY] Strategy switch: %s → %s", strategy.name(), next.name());
            strategy = next;
            currentTarget = Optional.empty();
        }
    }

    private void tickProduction(EnemyObservation obs, IntentQueue queue) {
        if (currentTarget.isEmpty()) {
            currentTarget = strategy.nextUnit(obs);
        }
        currentTarget.ifPresent(unit -> {
            int cost = SC2Data.mineralCost(unit);
            if (enemy.minerals >= cost) {
                enemy.minerals -= cost;
                String tag = "enemy-" + nextTag++;
                queue.add(new TrainIntent(tag, unit));
                currentTarget = Optional.empty();
                log.debugf("[ENEMY] Queued TrainIntent: %s (tag=%s)", unit, tag);
            }
        });
    }

    private void tickAttackLaunch(IntentQueue queue) {
        EnemyAttackConfig atk = strategy.attackConfig();
        boolean thresholdMet = enemy.stagingArea.size() >= atk.armyThreshold();
        boolean timerFired   = framesSinceLastAttack >= atk.attackIntervalFrames();
        if ((thresholdMet || timerFired) && !enemy.stagingArea.isEmpty()) {
            initialAttackSize = enemy.stagingArea.size();
            for (Unit u : enemy.stagingArea) {
                enemy.units.add(u);
                queue.add(new AttackIntent(u.tag(), NEXUS_POS));
            }
            log.infof("[ENEMY] Attack launched: %d units (threshold=%b timer=%b)",
                enemy.stagingArea.size(), thresholdMet, timerFired);
            enemy.stagingArea.clear();
            framesSinceLastAttack = 0;
        }
    }

    private void tickRetreat(IntentQueue queue) {
        if (initialAttackSize == 0) return;
        EnemyAttackConfig atk = strategy.attackConfig();

        // Per-unit HP retreat
        if (atk.retreatHealthPercent() > 0) {
            for (Unit u : enemy.units) {
                if (retreatingUnits.contains(u.tag())) continue;
                double totalHp  = u.health() + u.shields();
                double maxTotal = (double) u.maxHealth() + u.maxShields();
                if (totalHp / maxTotal * 100 < atk.retreatHealthPercent()) {
                    retreatingUnits.add(u.tag());
                    queue.add(new MoveIntent(u.tag(), STAGING_POS));
                    log.debugf("[ENEMY] Unit %s retreating (%.1f%% hp)", u.tag(),
                        totalHp / maxTotal * 100);
                }
            }
        }

        // Army-wide depletion retreat
        if (atk.retreatArmyPercent() > 0) {
            double survivingPct = (double) enemy.units.size() / initialAttackSize * 100;
            if (survivingPct < atk.retreatArmyPercent()) {
                for (Unit u : enemy.units) {
                    if (retreatingUnits.contains(u.tag())) continue;
                    retreatingUnits.add(u.tag());
                    queue.add(new MoveIntent(u.tag(), STAGING_POS));
                }
                log.infof("[ENEMY] Army retreat: %.0f%% surviving", survivingPct);
            }
        }

        // Transfer arrived retreating units to stagingArea
        enemy.units.removeIf(u -> {
            if (!retreatingUnits.contains(u.tag())) return false;
            if (EmulatedGame.distance(u.position(), STAGING_POS) >= 0.1) return false;
            retreatingUnits.remove(u.tag());
            enemy.stagingArea.add(u);
            return true;
        });
    }

    private Set<BuildingType> buildingTypes() {
        Set<BuildingType> types = new HashSet<>();
        for (Building b : enemy.buildings) types.add(b.type());
        return types;
    }

    public void reset(EnemyStrategy newStrategy) {
        this.strategy      = newStrategy;
        currentTarget      = Optional.empty();
        retreatingUnits.clear();
        initialAttackSize  = 0;
        framesSinceLastAttack = 0;
        nextTag            = 1000;
    }

    // Package-private accessors for EmulatedGame and tests
    Set<String>    retreatingUnits()                   { return Collections.unmodifiableSet(retreatingUnits); }
    void           setInitialAttackSizeForTesting(int n) { this.initialAttackSize = n; }
    EnemyStrategy  currentStrategy()                   { return strategy; }
}
```

```java
package io.quarkmind.sc2.emulated;

import io.quarkmind.domain.*;
import java.util.*;

public class ReactiveStrategy implements EnemyStrategy {

    private final int            reEvalInterval;
    private       EnemyStrategy  inner;
    private       String         pendingCounterName;

    public ReactiveStrategy(int reEvalInterval) {
        this.reEvalInterval = reEvalInterval;
        this.inner = EnemyStrategyLibrary.randomForRace(Race.PROTOSS); // initial default
    }

    @Override public String            name()            { return "REACTIVE"; }
    @Override public Race              race()            { return inner.race(); }
    @Override public int               mineralsPerTick() { return inner.mineralsPerTick(); }
    @Override public EnemyAttackConfig attackConfig()    { return inner.attackConfig(); }

    @Override
    public Optional<UnitType> nextUnit(EnemyObservation obs) {
        return inner.nextUnit(obs);
    }

    @Override
    public boolean shouldSwitch(EnemyObservation obs) {
        if (obs.gameFrame() % reEvalInterval != 0) return false;
        UnitType dominant = dominant(obs.playerUnits());
        pendingCounterName = counterFor(dominant);
        return true;
    }

    /** Called by EnemyBehavior when shouldSwitch() returns true — get the counter strategy. */
    public EnemyStrategy resolveCounter() {
        if (pendingCounterName == null) return inner;
        EnemyStrategy counter = EnemyStrategyLibrary.forName(pendingCounterName);
        inner = counter;
        pendingCounterName = null;
        return this; // ReactiveStrategy stays as the outer wrapper
    }

    String counterFor(UnitType dominant) {
        if (dominant == null) return randomName();
        return switch (dominant) {
            case STALKER, MARINE, MARAUDER, HYDRALISK, VOID_RAY -> randomRangedCounter();
            case ZEALOT, ZERGLING, BANELING                     -> "TERRAN_BIO_PUSH";
            case IMMORTAL, SIEGE_TANK, THOR, COLOSSUS           -> "ZERG_MASS_LING";
            default                                              -> randomName();
        };
    }

    private static UnitType dominant(List<Unit> units) {
        if (units.isEmpty()) return null;
        return units.stream()
            .collect(java.util.stream.Collectors.groupingBy(Unit::type,
                     java.util.stream.Collectors.counting()))
            .entrySet().stream()
            .max(Map.Entry.comparingByValue())
            .map(Map.Entry::getKey)
            .orElse(null);
    }

    private static String randomRangedCounter() {
        return new Random().nextBoolean() ? "PROTOSS_COLOSSUS_PUSH" : "ZERG_ROACH_RUSH";
    }

    private static String randomName() {
        List<String> names = new ArrayList<>(EnemyStrategyLibrary.allNames());
        names.remove("REACTIVE");
        return names.get(new Random().nextInt(names.size()));
    }

    @Override
    public void reset() { inner.reset(); pendingCounterName = null; }
}
```
